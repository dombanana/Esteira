Fechadura ainda nao instalada e contemplada: a porta existe no plano, fica

def roles_for(liberties: Iterable[str]) -> tuple[str, ...]:
    libs = set(liberties or ())
    out = [QUARTO, ENTRADA, SAIDA, ARMARIO]
    if "kitchen" in libs:
        out += [COZINHA, PANELAS]
    if "yard" in libs:

    return tuple(out)


def base_role(role: str) -> str:
    """v1.0.176: "ARMARIO+Q12B" (armario de outra cama do quarto alugado inteiro) -> "ARMARIO"."""
    return str(role).split("+", 1)[0]


def window(role: str, checkin: date, checkout: date, tz: ZoneInfo = SAO_PAULO) -> tuple[datetime, datetime]:
    (d0, h0, m0), (d1, h1, m1) = WINDOWS[base_role(role)]
    start = datetime.combine(checkin if d0 == "in" else checkout, time(h0, m0), tzinfo=tz)
    end = datetime.combine(checkin if d1 == "in" else checkout, time(h1, m1), tzinfo=tz)
    if end <= start:
        raise ValueError("reservation_window_invalid")
    return start.astimezone(timezone.utc), end.astimezone(timezone.utc)


def programada(role: str, checkin: date, checkout: date, tz: ZoneInfo = SAO_PAULO,
               env: Mapping[str, str] | None = None) -> tuple[datetime, datetime]:
    """A janela GRAVADA na fechadura. Igual a `window`, menos o inicio do QUARTO:
    `PMS_QUARTO_DESDE` ("11:30", Dom 30/09 -- contingencia) programa a senha antes; para o hospede o quarto
    continua sendo "a partir das 14h" (textos, botoes, luz). v1.0.159: o ARMARIO ja vale das 6h as 18h (como a entrada)."""
    import os
    starts, ends = window(role, checkin, checkout, tz)
    if base_role(role) != QUARTO:
        return starts, ends
    raw = str((os.environ if env is None else env).get("PMS_QUARTO_DESDE") or "").strip()
    m = re.fullmatch(r"(\d{1,2}):(\d{2})", raw)
    if not m or not (6 <= int(m.group(1)) < 14 and int(m.group(2)) < 60):
        return starts, ends
    cedo = datetime.combine(checkin, time(int(m.group(1)), int(m.group(2))), tzinfo=tz).astimezone(timezone.utc)
    return min(starts, cedo), ends


def revoke_groups() -> dict[tuple[int, int], tuple[str, ...]]:
    """Portas que param juntas (mesmo horario de fim) -> um comando de retirada."""
    groups: dict[tuple[int, int], list[str]] = {}
    for role in ROLES:
        _, (_, h, m) = WINDOWS[role]
        groups.setdefault((h, m), []).append(role)
    return {k: tuple(v) for k, v in sorted(groups.items())}


@dataclass
class LockMap:
    """Fechaduras da casa por papel. Quarto e armario por espaco (a cama sobe
    a arvore ate achar); as comuns, uma por casa."""

    rooms: dict[str, int] = field(default_factory=dict)
    armarios: dict[str, int] = field(default_factory=dict)
    common: dict[str, int] = field(default_factory=dict)

    @classmethod
    def parse(cls, raw: str) -> "LockMap":
        """`PMS_TTLOCK_LOCKS="Q06:15638346,ARMARIO_Q06:123,ENTRADA:16190236,SAIDA:...,PANELAS:...,QUINTAL:..."`."""
        out = cls()
        for item in (raw or "").split(","):
            if ":" not in item:
                continue
            key, value = (p.strip() for p in item.split(":", 1))
            if not key or not value.isdigit():
                continue
            key = key.upper()
            if key.startswith("ARMARIO_"):
                out.armarios[key[len("ARMARIO_"):]] = int(value)
            elif ALIASES.get(key, key) in COMMON_ROLES:
                out.common[ALIASES.get(key, key)] = int(value)
            else:
                out.rooms[key] = int(value)
        return out

    @classmethod
    def of(cls, rooms: Mapping[str, int] | None = None, common: Mapping[str, int] | None = None,
           armarios: Mapping[str, int] | None = None) -> "LockMap":
        return cls(dict(rooms or {}), dict(armarios or {}),
                   {ALIASES.get(k.upper(), k.upper()): int(v) for k, v in (common or {}).items()})


@dataclass(frozen=True, slots=True)
class Door:
    role: str
    lock_id: int | None          # None = fechadura ainda nao instalada
    starts: datetime
    ends: datetime


def door_plan(space: str, checkin: date, checkout: date, liberties: Iterable[str], *,
              room_lock: Callable[[str], int | None], armario_lock: Callable[[str], int | None],
              common: Mapping[str, int], tz: ZoneInfo = SAO_PAULO, extra_roles: Iterable[str] = ()) -> list[Door]:
    doors = []
    roles = list(roles_for(liberties))
    # porta comprada como adicional (ex.: cozinha completa do Hostel -> PANELAS), no horario da area comum
    roles += [r for r in extra_roles if r in COMMON_ROLES and r not in roles]
    for role in roles:
        if role == QUARTO:
            lock = room_lock(space)
        elif role == ARMARIO:
            lock = armario_lock(space)
        else:
            lock = common.get(role)
        starts, ends = programada(role, checkin, checkout, tz)
        doors.append(Door(role, lock, starts, ends))
    return doors

def room_prefix(space: str) -> str:
    """"06" -> "06"; "Q06" -> "06"; "Q06-B" ou "Q06B" (cama B do 06) -> "62" (regra da V3.4)."""
    match = re.fullmatch(r"Q?(\d{2})(?:-?([A-F]))?", space.strip().upper())
    if not match:
        raise ValueError("space_code_invalid")
    room, bed = int(match.group(1)), match.group(2)
    return f"{room % 10}{ord(bed) - 64}" if bed else f"{room:02d}"


def digits_problem(value: str) -> str | None:
    """None se os 4 digitos servem; senao, a frase para o hospede."""
    d = str(value or "").strip()
    if not re.fullmatch(r"\d{4}", d):
        return "A senha pessoal tem exatamente 4 numeros."
    if len(set(d)) == 1:
        return "Escolha 4 numeros que nao sejam todos iguais (como 1111)."
    steps = {int(d[i + 1]) - int(d[i]) for i in range(3)}
    if steps in ({1}, {-1}):
        return "Escolha 4 numeros que nao sejam uma sequencia (como 1234 ou 4321)."
    return None


def suggestions(phone: str | None, document: str | None) -> list[str]:
    """Os 4 ultimos do telefone e os 4 primeiros do documento -- so os que servem."""
    out = []
    p = re.sub(r"\D", "", str(phone or ""))
    n = re.sub(r"\D", "", str(document or ""))
    for cand in ((p[-4:] if len(p) >= 4 else ""), (n[:4] if len(n) >= 4 else "")):
        if cand and digits_problem(cand) is None and cand not in out:
            out.append(cand)
    return out

STAFF_PREFIX, COWORKING_PREFIX, HOUSE_PASS_PREFIX = "00", "88", "99"
COWORKING_KEYS = frozenset({"coworking", "cowork", "cw"})
HOUSE_PASS_KEYS = frozenset({"house_pass", "house-pass", "housepass", "hp"})


def access_prefix(space: str, instance: str = "") -> str:
    """Os 2 primeiros digitos da senha: o quarto/cama, ou 88 (coworking) / 99 (House Pass).
    A equipe (00) nao tem Evento de estadia: a senha dela sai da governanca."""
    for key in (str(instance or ""), str(space or "")):
        k = key.strip().lower()
        if k in COWORKING_KEYS:
            return COWORKING_PREFIX
        if k in HOUSE_PASS_KEYS:
            return HOUSE_PASS_PREFIX
    return room_prefix(space)

def all_prefixes(spaces: Any = ()) -> set[str]:

def contains_staff_code(guest_digits: str, staff_codes: Any, prefixes: Any) -> bool:
    """A fechadura aceita a senha certa no meio de outros numeros. Se a senha de um hospede (prefixo + 4
    digitos) tiver, em sequencia, os 4 numeros de alguem da equipe, ela abriria as portas da equipe."""
    codes = [str(c) for c in staff_codes if c]
    return any(c in (p + str(guest_digits)) for p in prefixes for c in codes)


def guest_pin(space: str, digits: str, instance: str = "") -> str:

def pin_options(*, cpf: str | None = None, birth_date: str | None = None, phone: str | None = None) -> list[dict]:
    """As opcoes de senha que servem para esta pessoa: 4 primeiros do CPF, dia e mes do nascimento (DDMM), 4
    ultimos do telefone. So entram as que passam na regra (sem sequencia, sem repeticao)."""
    out: list[dict] = []
    c = re.sub(r"\D", "", str(cpf or ""))
    if len(c) == 11:
        out.append({"kind": "cpf", "digits": c[:4]})
    m = re.fullmatch(r"(\d{4})-(\d{2})-(\d{2})", str(birth_date or "")[:10])
    if m:
        out.append({"kind": "nascimento", "digits": m.group(3) + m.group(2)})
    p = re.sub(r"\D", "", str(phone or ""))
    if len(p) >= 8:
        out.append({"kind": "telefone", "digits": p[-4:]})
    return [o for o in out if digits_problem(o["digits"]) is None]

def random_digits(taken: Any = (), rng: Any = None, alfabeto: str = "0123456789") -> str:
    """4 numeros aleatorios validos, fora dos ja usados (sugestao quando a opcao escolhida repete)."""
    import secrets as _s
    if alfabeto != "0123456789":
        pick = (lambda: "".join(rng.choice(alfabeto) for _ in range(4))) if rng is not None else \
               (lambda: "".join(_s.choice(alfabeto) for _ in range(4)))
    else:
        pick = (lambda: f"{rng.randrange(10000):04d}") if rng is not None else (lambda: f"{_s.randbelow(10000):04d}")
    used = set(taken or ())
    for _ in range(500):
        d = pick()
        if digits_problem(d) is None and d not in used:
            return d
    raise ValueError("sem 4 digitos livres")

def access_plan(store: Any, outbox: Any, *, now: datetime) -> list[RoomPlan]:

def receive(state: Any, payload: Any, *, env: Mapping[str, str] = os.environ, at: datetime | None = None,

    medidas: list[dict] = []
    chegou = False
    for op in openings(payload):
        if _door_of(locks, op.lock_id) not in cfg.doors:
