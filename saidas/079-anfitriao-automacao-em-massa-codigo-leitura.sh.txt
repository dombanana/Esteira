def catalogo(state: Any, env: Mapping[str, str] = os.environ) -> dict[str, dict]:
    """chave -> {tipo, nome, espaco, regra|ap|aparelho}. Um aparelho aparece uma vez so."""

def ligado(state: Any, item: Mapping[str, Any], at: datetime) -> bool | None:
    from app.tuya_casa import service as S
    from app.tuya_casa import ventilador as V
    if item["tipo"] == "ar":
        regra = item["regra"]
        row = (getattr(state.store, "tuya_clima", {}) or {}).get(regra.espaco or regra.aparelho) or {}
        return S.ar_ligado(state, regra, row, at)
    if "ap" in item:

    ap = item["aparelho"]

def ordem(store: Any, chave: str, at: datetime) -> dict | None:
    """A ordem do anfitriao que vale agora para este aparelho (ou None)."""
    o = (getattr(store, "automacao_ordens", {}) or {}).get(chave)
    if not o or o.get("solta"):
        return None
    if o.get("ate") and str(o["ate"]) <= at.isoformat():
        return None
    return o


def manda(store: Any, chave: str, at: datetime) -> bool:
    """O vigia do quarto (ar/ventilador) pergunta: o anfitriao esta mandando neste aparelho agora?"""

def painel(state: Any, at: datetime, env: Mapping[str, str] = os.environ) -> dict:
    tz = ZoneInfo(state.store.timezone)
    itens = []
    for chave, it in catalogo(state, env).items():

                      "ligado": ligado(state, it, at), "temp": temperatura(state, it, env),
                      "ocupado": ocupado(state, it, at),
                      "ordem": None if o is None else {"ligar": o["ligar"], "origem": o.get("origem", ""),
                                                        "ate": datetime.fromisoformat(o["ate"]).astimezone(tz).strftime("%d/%m %H:%M") if o.get("ate") else ""}})
    regras = sorted((getattr(state.store, "automacao_regras", {}) or {}).values(), key=lambda r: r.get("nome", ""))
    return {"aparelhos": itens, "regras": [r for r in regras if not r.get("removida")]}

def _gravar_ordem(ctx: IntentContext, chave: str, novo: dict | None) -> None:
    store = ctx.params["store"]
    antes = (getattr(store, "automacao_ordens", {}) or {}).get(chave)
    if novo is None:
        if not antes or antes.get("solta"):
            return
        novo = {**antes, "solta": ctx.at.isoformat()}
    ctx.change("automacao_ordem", chave, antes, novo)
    ctx.transaction.after_commit(lambda: store.automacao_ordens.__setitem__(chave, novo))


def _mandar(ctx: IntentContext) -> str:
    chaves = [str(c) for c in ctx.params.get("chaves") or () if str(c)]
    if not chaves:
        raise ValueError("Escolha pelo menos um aparelho.")
    ligar = bool(ctx.params.get("ligar"))
    horas = ctx.params.get("horas")
    ate = ""
    if horas not in (None, "", 0, "0"):
        try:
            h = float(horas)
        except (TypeError, ValueError):
            raise ValueError("Horas inválidas.") from None
        if not 0 < h <= 72:
            raise ValueError("De 0 a 72 horas.")
        ate = (ctx.at + timedelta(hours=h)).isoformat()
    for c in chaves:
        _gravar_ordem(ctx, c, {"ligar": ligar, "ate": ate, "por": ctx.caller.actor.id, "at": ctx.at.isoformat(),
                               "origem": "agora"})
    return f"{len(chaves)} aparelho(s): {'ligando' if ligar else 'desligando'}" + (f" por {horas} h." if ate else " até você soltar.")


def _soltar(ctx: IntentContext) -> str:
    chaves = [str(c) for c in ctx.params.get("chaves") or () if str(c)]
    for c in chaves:
        _gravar_ordem(ctx, c, None)
    return f"{len(chaves)} aparelho(s) de volta à regra do quarto."


def _hora(txt: Any) -> str:
    try:

        return dtime(int(h), int(m or 0)).strftime("%H:%M")
    except (TypeError, ValueError):
        return ""
