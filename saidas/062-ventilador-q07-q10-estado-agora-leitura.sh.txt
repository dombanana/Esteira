def temperatura(status: Mapping[str, Any], codigo: str = "", escala: float = 0.0) -> float | None:
    """Graus Celsius do status Tuya (a maioria manda x10: 253 = 25,3). `codigo`/`escala`: o termometro que a Tuya nao
    descreve (controle infravermelho: "dp_101", escala 10)."""
    for code in ((codigo,) if codigo else TEMP_CODES):
        if code in status:
            try:
                v = float(status[code])
            except (TypeError, ValueError):
                return None
            if escala:
                return v / escala
            return v / 10 if abs(v) > 60 else v
    return None
