"""Desconto do plano Smart (regra do Dom, 24/09/2026): "todas as pessoas que forem plano
Smart tem 10% de desconto em tudo que consumirem na casa".

Um ponto so, usado por tudo o que o hospede compra da casa:
  * adicionais do link (amenities, bicicleta, cozinha, late checkout...) -- `catalog._order`;
  * pacotes (aberturas a distancia, mensagens da Casa Viva) -- `packs.order`;
  * adicional lancado pela equipe na reserva -- `intents._add_extra`;
  * lojinha / honest market -- `shop._checkout`.

Fora do desconto: credito comprado para a Carteira (nao e consumo; o desconto ja vale
quando o credito e usado) e a propria diaria (a hospedagem nao e "consumo na casa").

Lojinha (PROPOSTA, a confirmar com o Dom): quem paga o desconto e a CASA, nao o lojista.
O lojista recebe pelo preco cheio; a comissao da casa cai no valor do desconto.

`PMS_DESCONTO_SMART` muda a porcentagem (0 desliga); padrao 0 desde a v1.0.147 (Dom 24/10: "nao tem mais os 10%
de desconto do pacote Smart... nao contabiliza mais 10% de desconto para nada"). O mecanismo fica: ligar e decisao da casa.
"""
from __future__ import annotations

import os
from typing import Any, Mapping

DEFAULT_PCT = 0


def pct(env: Mapping[str, str] | None = None) -> int:
    raw = str((env if env is not None else os.environ).get("PMS_DESCONTO_SMART", DEFAULT_PCT)).strip()
    try:
        return max(0, min(int(raw), 100))
    except ValueError:
        return DEFAULT_PCT


def is_smart(event: Any) -> bool:
    """Smart = a estadia tem a cozinha do quarto (mesma regra de `catalog` e da Casa Viva)."""
    if event is None:
        return False
    return any(getattr(l, "value", l) == "kitchen" for l in event.scope.liberties)


def pct_for(event: Any, env: Mapping[str, str] | None = None) -> int:
    return pct(env) if is_smart(event) else 0


def apply(cents: int, percent: int) -> tuple[int, int]:
    """(valor a pagar, desconto) em centavos. Meio centavo arredonda para cima, igual a tela."""
    cents = int(cents)
    off = (cents * percent + 50) // 100 if percent else 0      # meio centavo para cima (igual ao Math.round da tela)
    return cents - off, off


def priced(row: dict, event: Any, list_cents: int) -> dict:
    """Linha de adicional com o desconto aplicado: `amount_cents` ja e o que se paga;
    `list_cents` e `discount_cents` ficam para o hospede ver e para a conta."""
    percent = pct_for(event)
    net, off = apply(list_cents, percent)
    out = {**row, "amount_cents": net}
    if off:
        out.update({"list_cents": int(list_cents), "discount_cents": off, "discount": f"smart-{percent}"})
    return out
