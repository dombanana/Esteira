O MESMO desenho e a MESMA regra do modulo de Automacao planejado na V1.1 (simulacao aprovada pelo Dom em 27/10):
  "E um ventilador, nao um ar-condicionado: ele mexe o ar do quarto, nao esfria."
  Quando ligar:
    manual       o hospede liga e desliga pela chave;
    temperatura  liga sozinho quando o quarto passa 1 grau do alvo e desliga quando volta ao alvo (zona morta: fica);
    horario      liga e desliga em horas marcadas (pode atravessar a meia-noite).
  Velocidade: 1, 2, 3 ou auto (auto escolhe pela distancia ate o alvo).
  Tocar a chave no automatico passa para o manual.

Os aparelhos entram pelo NOME na Tuya (app/luz/auto.py): "Ventilador 02", "Termometro 02". Sem aparelho no quarto, o
bloco nao aparece. O quarto vazio (fim da estadia) desliga e a escolha volta ao comeco para o proximo hospede.

v1.0.265 (Dom 18/11, ventilador do quarto 8 -- tudo aprovado: "Pode montar essa versao em alta, ta tudo aprovado ai"):
  * modulo de 3 reles, um por velocidade (fraco, medio, forte): "nunca 2 botoes acionados ao mesmo tempo. Quando aciona
    o botao forte, o fraco e o medio tem que estar desligado" -- toda troca DESLIGA os outros, espera 1 s e so entao liga

def vel_auto(temp: float | None, alvo: float | None, atual: Any = None) -> int:
    """v1.0.265: uma velocidade a cada grau acima da regua (alvo+1 fraco, +2 medio, +3 forte); desce meio grau abaixo."""
    if temp is None or alvo is None:
        return 2
    d = temp - alvo
    base = 1 if d < 2 else 2 if d < 3 else 3
    try:
        a = int(atual or 0)
    except (TypeError, ValueError):
        a = 0
    if 3 >= a > base and d >= a - 0.5:
        return a
    return base


def _hora(txt: Any) -> time | None:
    try:
        h, m = str(txt or "").strip().replace("h", ":").split(":")[:2]
        return time(int(h), int(m or 0))
    except (ValueError, TypeError):
        return None


def no_intervalo(agora: time, ini: time, fim: time) -> bool:
    return ini <= agora < fim if ini < fim else (agora >= ini or agora < fim)


def decidir(modo: str, *, alvo: float | None, liga: str | None, desliga: str | None, velocidade: str,
            manual_ligado: bool, temp: float | None, agora_t: time, ligado_atual: bool,
            vel_atual: Any = None) -> tuple[bool, int | None]:
    if modo == "manual":
        ligado = bool(manual_ligado)
    elif modo == "temperatura":
        if temp is None or alvo is None:
            ligado = ligado_atual                 # sem leitura: nao mexe
        elif temp >= alvo + 1:
            ligado = True
        elif temp <= alvo:
            ligado = False
        else:
            ligado = ligado_atual                 # zona morta: evita liga-desliga
    elif modo == "horario":
        a, b = _hora(liga), _hora(desliga)
        ligado = bool(a and b) and no_intervalo(agora_t, a, b)
    else:
        raise ValueError("modo desconhecido")
    if not ligado:
        return False, None
    base = alvo if alvo is not None else (FAIXA[0] + FAIXA[1]) / 2        # v1.0.265: manual + auto = meio da regua
    vel = vel_auto(temp, base, vel_atual) if velocidade == "auto" else int(velocidade)
    return True, vel
