def _quarto(espaco: str) -> str:
    s = str(espaco or "").strip().upper()
    if not s:
        return ""
    if not s.startswith("Q"):
        s = "Q" + s
    return s if len(s) >= 3 else s[0] + s[1:].zfill(2)


def tem_vel(ap: Mapping[str, Any]) -> bool:
    return bool(ap.get("vel_code") or ap.get("reles"))


def _vel_n(vel: Any) -> int:
    try:
        return max(1, min(3, int(vel or VEL_INICIAL)))
    except (TypeError, ValueError):
        return int(VEL_INICIAL)


def pedido(ap: Mapping[str, Any], ligar: bool, vel: Any) -> dict:
    """O que vai para a Tuya. Reles (v1.0.265): desligar = os 3 desligados; ligar na velocidade n = primeiro os OUTROS
    desligados, espera PASSO_S e so entao o rele n ligado (`depois_so_se_ok`: se o primeiro falhar, o segundo nao sai)."""
    reles = list(ap.get("reles") or [])
    if not reles:
        return {"commands": comandos(ap, ligar, vel)}
    if not ligar:
        return {"commands": [{"code": r, "value": False} for r in reles]}
    escolhido = reles[_vel_n(vel) - 1]
    return {"commands": [{"code": r, "value": False} for r in reles if r != escolhido],
            "depois": [{"code": escolhido, "value": True}], "pulso_s": PASSO_S, "depois_so_se_ok": True}


def comandos(ap: Mapping[str, Any], ligar: bool, vel: int | None) -> list[dict]:
    if ap.get("reles"):                                    # v1.0.265: a ordem toda (o envio separa em dois passos)
        p = pedido(ap, ligar, vel)
        return list(p["commands"]) + list(p.get("depois") or [])
    if not ligar:
        return [{"code": ap["switch"], "value": False}]
    out = [{"code": ap["switch"], "value": True}]
    vals = ap.get("vel_vals") or []
    if ap.get("vel_code") and vel and len(vals) >= 3:
        out.append({"code": ap["vel_code"], "value": vals[max(1, min(3, int(vel))) - 1]})
    return out


def _ligado(v: Any) -> bool:
    return v in (True, 1, "1", "true", "on")


def reles_ligados(state: Any, ap: Mapping[str, Any]) -> list[int] | None:
    """v1.0.265: quais velocidades (1, 2, 3) estao ligadas agora no modulo de reles; None sem leitura."""
    from app.tuya_casa.service import ultimo
    st = (ultimo(state).get(ap["device"]) or {}).get("status") or {}
    reles = list(ap.get("reles") or [])
    if not reles or not any(r in st for r in reles):
        return None
    return [n for n, r in enumerate(reles, 1) if _ligado(st.get(r))]

def vel_real(state: Any, ap: Mapping[str, Any]) -> int | None:
    """A velocidade ligada agora (so modulo de reles e so quando ha exatamente uma ligada)."""
    on = reles_ligados(state, ap)

def estado_real(state: Any, ap: Mapping[str, Any]) -> bool | None:
    from app.tuya_casa.service import ultimo
    if ap.get("reles"):
        on = reles_ligados(state, ap)
        return None if on is None else bool(on)
    st = (ultimo(state).get(ap["device"]) or {}).get("status") or {}
    if ap["switch"] in st:
        return st[ap["switch"]] in (True, 1, "1", "true", "on")
    return None

    from app.tuya_casa.service import temperatura as temp_de, ultimo

def hospede_do_quarto(store: Any, quarto: str, at: datetime) -> Any:
    """A estadia valendo agora no quarto. v1.0.27x (Dom 18/11, sobre o ar: "a pessoa fez o check-out, ja deu o horario
    do check-out, nao tem ninguem mais para entrar, ai ele desliga" -- e para o ventilador: "Ok, mesma regra"): a janela
    do QUARTO, a mesma do ar (14h da chegada -> hora da saida; entrada antecipada paga a partir das 6h), nao mais a da
    porta de entrada (6h -> 18h)."""
    from zoneinfo import ZoneInfo

    from app.core.event import EventState
    from app.tuya_casa.service import _janela_do_ar
    tz = ZoneInfo(store.timezone)
    for e in store.events.values():
        if e.state not in (EventState.CONFIRMED, EventState.IN_PROGRESS) or not e.where or e.where[0] != quarto:
            continue
        try:
            ini, fim = _janela_do_ar(store, e, tz)
        except ValueError:
            continue
        if ini <= at < fim:
            return e
    return None


def linha(store: Any, quarto: str, event_id: str) -> dict:
    """A escolha do quarto PARA esta estadia (a do hospede anterior nao vale)."""
    row = dict((getattr(store, "tuya_vent", {}) or {}).get(quarto) or {})
    if row.get("event_id") != event_id:
        row = {"modo": "manual", "alvo": None, "liga": None, "desliga": None, "velocidade": VEL_INICIAL,
               "manual_ligado": False, "event_id": event_id}
    return row


def _desde(row: Mapping[str, Any], at: datetime) -> float:
    try:
        return (at - datetime.fromisoformat(str(row.get("enviado_at")))).total_seconds()
    except (TypeError, ValueError):
        return 1e9
