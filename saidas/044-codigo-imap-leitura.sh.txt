"""v1.0.243 (Dom 14/11, "Alta": a entrada das reservas sem a Nobeds): a caixa das reservas lida direto por IMAP, so leitura;
os e-mails de OTA entram pela mesma porta do webhook (modo sombra), uma vez so; sem credencial, nada roda."""
from __future__ import annotations

from email.message import EmailMessage

from app.entrada import imap_reservas as IM
from app.entrada.tests.test_porta_e_motor import _airbnb
from app.finance.tests.test_direct_booking import AGORA, _casa

ENV = {"PMS_RESERVAS_IMAP_USUARIO": "contato@exemplo.com", "PMS_RESERVAS_IMAP_SENHA": "x" * 12}


def _msg(mid, remetente, assunto, texto):
    m = EmailMessage()
    m["From"], m["To"], m["Subject"], m["Message-ID"], m["Date"] = remetente, "contato@exemplo.com", assunto, mid, \
        "Sat, 14 Nov 2026 13:30:16 +0000"
    m.set_content(texto)
    return m.as_bytes()


class Caixa:
    """IMAP falso: guarda o que foi pedido para conferir que tudo e so leitura."""

    def __init__(self, msgs):
        self.msgs, self.pedidos, self.select_ro = msgs, [], []

    def login(self, u, s):
        self.pedidos.append("login")

    def select(self, pasta, readonly=False):
        self.select_ro.append(readonly)
        return ("OK", [b"1"]) if "03_Reservas" in pasta else ("NO", [b""])

    def search(self, charset, crit):
        return "OK", [b" ".join(str(i + 1).encode() for i in range(len(self.msgs)))]

    def fetch(self, num, partes):
        self.pedidos.append(partes)
        assert "PEEK" in partes                                       # nunca marca como lido
        return "OK", [(b"x", self.msgs[int(num) - 1])]

    def logout(self):
        pass


def test_sem_credencial_nada_roda():
    state = _casa()
    assert IM.config({}) is None and IM.rodar(state, env={}) == ""


def test_email_do_airbnb_entra_pela_porta_casado_e_uma_vez_so():
    state = _casa()
    _airbnb(state, code="HMABC12345")
    caixa = Caixa([_msg("<a1@airbnb.com>", "Airbnb <automated@airbnb.com>",
                        "Reserva confirmada - Ana chega em 20 de nov.", "Codigo de confirmacao HMABC12345"),
                   _msg("<n1@x.com>", "Loja <ofertas@loja.com>", "Promocao", "nada")])
    IM._ULTIMA.clear(); IM._VISTOS.clear()
    out = IM.rodar(state, env=ENV, conectar=lambda host: caixa, at=AGORA)
    assert "1 novo" in out
    rows = [x for x in state.store.entradas.values() if x.get("ota") == "Airbnb"]
    assert len(rows) == 1 and rows[0]["codigo"] == "HMABC12345" and rows[0]["reservation_id"]
    assert rows[0]["tipo"] == "reserva_nova" and rows[0]["autoria"] == "canal:imap" and rows[0]["modo"] == "sombra"
    assert all(caixa.select_ro)                                       # a caixa so e aberta para leitura
    assert not [x for x in state.store.entradas.values() if x.get("dominio") == "loja.com"]   # so OTA
    IM._ULTIMA.clear()
    assert IM.rodar(state, env=ENV, conectar=lambda host: caixa, at=AGORA) == ""            # uma vez so
