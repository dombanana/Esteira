def _texto(msg: email.message.Message) -> str:
    """O texto do e-mail: a parte text/plain; sem ela, o HTML (a porta de entrada tira as tags)."""
    plano, html = "", ""
    partes = msg.walk() if msg.is_multipart() else [msg]
    for p in partes:
        tipo = p.get_content_type()
        if p.get_content_maintype() == "multipart" or "attachment" in str(p.get("Content-Disposition") or ""):
            continue
        try:
            corpo = p.get_payload(decode=True) or b""
            t = corpo.decode(p.get_content_charset() or "utf-8", "replace")
        except Exception:  # noqa: BLE001
            continue
        if tipo == "text/plain" and not plano:
            plano = t
        elif tipo == "text/html" and not html:
            html = t
    return plano or html


def para_porta(bruto: bytes) -> dict:
    """O e-mail cru no formato da porta de entrada (`entrada.ler_email`)."""
    msg = email.message_from_bytes(bruto)
    remetente = parseaddr(_dec(msg.get("From")))[1].lower()
    para = parseaddr(_dec(msg.get("Delivered-To") or msg.get("To")))[1].lower()
    return {"message_id": str(msg.get("Message-ID") or "").strip()[:200], "remetente": remetente,
            "dominio": remetente.rsplit("@", 1)[-1] if "@" in remetente else "", "para": para,
            "assunto": _dec(msg.get("Subject"))[:300], "texto": _texto(msg), "quando": str(msg.get("Date") or "")[:40]}
