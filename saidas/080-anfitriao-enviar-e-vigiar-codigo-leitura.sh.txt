def _regra(ctx: IntentContext) -> str:
    store = ctx.params["store"]
    r = dict(ctx.params.get("regra") or {})
    rid = str(r.get("id") or "") or "regra-" + hashlib.sha256(f"{ctx.caller.actor.id}|{ctx.at.isoformat()}".encode()).hexdigest()[:10]
    antes = (getattr(store, "automacao_regras", {}) or {}).get(rid)
    if r.get("remover"):
        if not antes:
            raise ValueError("Programação não encontrada.")
        novo = {**antes, "removida": ctx.at.isoformat(), "ativa": False}
    else:
        tipo = str(r.get("tipo") or "")
        if tipo not in ("horario", "temperatura"):
            raise ValueError("Escolha: por horário ou por temperatura.")
        chaves = [str(c) for c in r.get("chaves") or () if str(c)]
        if not chaves:
            raise ValueError("Escolha os aparelhos.")
        liga, desliga = _hora(r.get("liga")), _hora(r.get("desliga"))
        dias = [d for d in (r.get("dias") or DIAS) if d in DIAS] or list(DIAS)
        novo = {"id": rid, "nome": str(r.get("nome") or "Programação")[:40], "tipo": tipo, "chaves": chaves,
                "liga": liga, "desliga": desliga, "dias": dias,
                "ocupados": "incluir" if r.get("ocupados") == "incluir" else "pular",
                "ativa": r.get("ativa", True) is not False, "por": ctx.caller.actor.id, "at": ctx.at.isoformat()}
        if tipo == "horario":
            if not (liga and desliga) or liga == desliga:
                raise ValueError("Escolha a hora de ligar e a de desligar (diferentes).")
        else:
            try:
                acima = float(str(r.get("acima")).replace(",", "."))
                abaixo = float(str(r.get("abaixo") if r.get("abaixo") not in (None, "") else acima - 1).replace(",", "."))
            except (TypeError, ValueError):
                raise ValueError("Temperatura inválida.") from None
            if not (10 <= abaixo < acima <= 45):
                raise ValueError("Desliga abaixo de quanto? Tem que ser menor que o liga (ex.: liga 28°, desliga 26°).")
            novo.update({"acima": acima, "abaixo": abaixo})
            if bool(liga) != bool(desliga):
                raise ValueError("Horário da temperatura: preencha os dois ou nenhum.")
    ctx.change("automacao_regra", rid, antes, novo)
    ctx.transaction.after_commit(lambda: store.automacao_regras.__setitem__(rid, novo))
    if novo.get("removida") or not novo.get("ativa"):   # a regra que sai solta o que ela mesma mandou
        for c, o in list((getattr(store, "automacao_ordens", {}) or {}).items()):
            if o.get("origem") == rid and not o.get("solta"):
                _gravar_ordem(ctx, c, None)
    return f"Programação {'apagada' if novo.get('removida') else 'guardada'}: {novo.get('nome')}."


def _aplicar(ctx: IntentContext) -> str:
    feitos = 0
    for c, novo in ctx.params.get("ordens") or ():
        _gravar_ordem(ctx, c, novo)
        feitos += 1
    return f"{feitos} ordem(ns) das programações."


def register(router: IntentRouter) -> None:

def _na_janela(regra: Mapping[str, Any], agora: datetime) -> bool:
    liga, desliga = regra.get("liga"), regra.get("desliga")
    if not liga or not desliga:
        return DIAS[agora.weekday()] in regra.get("dias", DIAS)
    t = agora.strftime("%H:%M")
    if liga < desliga:
        return liga <= t < desliga and DIAS[agora.weekday()] in regra.get("dias", DIAS)
    # atravessa a meia-noite: o dia que vale e o do comeco
    if t >= liga:
        return DIAS[agora.weekday()] in regra.get("dias", DIAS)
    if t < desliga:
        return DIAS[(agora - timedelta(days=1)).weekday()] in regra.get("dias", DIAS)
    return False


def planejar(state: Any, at: datetime, env: Mapping[str, str] = os.environ) -> list[tuple[str, dict | None]]:
    """As ordens que as regras pedem agora: [(chave, ordem nova | None = soltar)]. Puro (le, nao grava)."""
    store = state.store
    agora = at.astimezone(ZoneInfo(store.timezone))
    cat = catalogo(state, env)
    ordens = getattr(store, "automacao_ordens", {}) or {}
    out: list[tuple[str, dict | None]] = []
    for r in (getattr(store, "automacao_regras", {}) or {}).values():
        if not r.get("ativa") or r.get("removida"):
            continue
        janela = _na_janela(r, agora)
        for c in r.get("chaves") or ():
            it = cat.get(c)
            if it is None:
                continue
            atual = ordens.get(c) or {}
            minha = atual.get("origem") == r["id"] and not atual.get("solta")
            if atual and not atual.get("solta") and not minha and ordem(store, c, at):
                continue                               # outra ordem (do anfitriao ou de outra regra) manda
            if r.get("ocupados") != "incluir" and ocupado(state, it, at):
                if minha:
                    out.append((c, None))              # o quarto ficou ocupado: devolve ao hospede
                continue
            base = {"por": r.get("por", ""), "at": at.isoformat(), "origem": r["id"], "ate": ""}
            if r["tipo"] == "horario":
                if janela and not (minha and atual.get("ligar")):
                    out.append((c, {**base, "ligar": True}))
                elif not janela and minha and atual.get("ligar"):
                    out.append((c, {**base, "ligar": False, "ate": (at + timedelta(minutes=3)).isoformat()}))
            else:
                t = temperatura(state, it, env)
                if t is None:
                    continue
                if janela and t >= r["acima"] and not (minha and atual.get("ligar")):
                    out.append((c, {**base, "ligar": True}))
                elif minha and atual.get("ligar") and (t <= r["abaixo"] or not janela):
                    out.append((c, {**base, "ligar": False, "ate": (at + timedelta(minutes=3)).isoformat()}))
    return out
