class TtlockClient:
    def __init__(self, base_url: str, client_id: str, client_secret: str, username: str, password_md5: str, timeout: int = 30):
        if not all(str(v).strip() for v in (client_id, client_secret, username, password_md5)):
            raise ValueError("ttlock_credentials_required")
        self.base_url = base_url.rstrip("/")
        self.client_id, self.client_secret = client_id, client_secret
        self.username, self.password_md5 = username, password_md5
        self.timeout = timeout
        self._access_token: str | None = None

    @classmethod
    def from_environment(cls) -> "TtlockClient":
        import os
        return cls(
            os.environ.get("TTLOCK_API_BASE_URL", "https://api.sciener.com"),
            os.environ["TTLOCK_CLIENT_ID"], os.environ["TTLOCK_CLIENT_SECRET"],
            os.environ["TTLOCK_USERNAME"], os.environ["TTLOCK_PASSWORD_MD5"],
        )

    def _post(self, path: str, data: Mapping[str, Any]) -> Mapping[str, Any]:
        request = urllib.request.Request(
            self.base_url + path, method="POST",
            data=urllib.parse.urlencode({k: str(v) for k, v in data.items()}).encode("ascii"),
            headers={"Content-Type": "application/x-www-form-urlencoded"},
        )
        try:
            with urllib.request.urlopen(request, timeout=self.timeout) as response:
                payload = json.loads(response.read().decode("utf-8") or "{}")
        except urllib.error.HTTPError as exc:
            raise RuntimeError(f"ttlock_http_{exc.code}") from None
        except (urllib.error.URLError, OSError):
            raise RuntimeError("ttlock_transport") from None
        code = payload.get("errcode") if isinstance(payload, Mapping) else None
        if code not in (None, 0, "0"):
            text = re.sub(r"[^A-Za-z0-9 _.,:-]", "", str(payload.get("errmsg") or ""))[:80]
            raise RuntimeError(f"ttlock_provider_{code}" + (f":{text}" if text else ""))
        return payload

    def authenticate(self) -> str:
        payload = self._post("/oauth2/token", {
            "client_id": self.client_id, "client_secret": self.client_secret,
            "username": self.username, "password": self.password_md5,
        })
        token = str(payload.get("access_token") or "")
        if not token:
            raise RuntimeError("ttlock_token_missing")
        self._access_token = token
        return token

    def _authorized(self, path: str, data: Mapping[str, Any]) -> Mapping[str, Any]:
        token = self._access_token or self.authenticate()
        return self._post(path, {"clientId": self.client_id, "accessToken": token,
                                 "date": int(time.time() * 1000), **data})

    def add_custom_passcode(self, lock_id: int, pin: str, name: str, starts_at: datetime, ends_at: datetime) -> str:
        if ends_at <= starts_at:
            raise ValueError("ttlock_window_invalid")
        payload = self._authorized("/v3/keyboardPwd/add", {
            "lockId": lock_id, "keyboardPwd": pin, "keyboardPwdName": name[:40],
            "startDate": int(starts_at.timestamp() * 1000), "endDate": int(ends_at.timestamp() * 1000),
            "addType": 2,
        })
        return str(payload.get("keyboardPwdId") or "")

    def delete_passcode(self, lock_id: int, keyboard_pwd_id: str) -> None:
        self._authorized("/v3/keyboardPwd/delete", {"lockId": lock_id, "keyboardPwdId": keyboard_pwd_id, "deleteType": 2})

        `changeType=2` = pelo gateway (a fechadura recebe sem ninguem ao lado). A senha em si nao muda."""

            "lockId": lock_id, "keyboardPwdId": keyboard_pwd_id,
            "startDate": int(starts_at.timestamp() * 1000), "endDate": int(ends_at.timestamp() * 1000),

    def list_locks(self) -> list:
        """Todas as fechaduras da conta (varredura do cadastro da casa; so leitura)."""
        out, page = [], 1
        while page <= 20:
            payload = self._authorized("/v3/lock/list", {"pageNo": page, "pageSize": 100})

            page += 1

class CredentialStore:
    """Credenciais emitidas, por Evento, em memoria (testes). O PIN em claro
    NUNCA fica aqui -- so o id no TTLock e uma impressao digital."""

    by_event: dict[str, dict[str, str]] = field(default_factory=dict)

    def get(self, event_id: str) -> dict[str, str] | None:
        return self.by_event.get(event_id)

    def set(self, event_id: str, data: dict[str, str]) -> None:

    def clear(self, event_id: str) -> None:

class VaultCredentials:
    """As mesmas credenciais, no cofre (A13): sobrevivem ao reinicio do worker,
    entao uma senha nunca e emitida duas vezes nem esquecida na revogacao.
    Cada leitura fica registrada como acesso ao cofre."""

    def __init__(self, vault: Vault) -> None:
        self.vault = vault

    def get(self, event_id: str) -> dict[str, str] | None:
        try:
            data = dict(self.vault.open(f"cred:{event_id}", actor="adapter:ttlock", purpose="credential"))
        except VaultMiss:
            return None
        return None if data.get("revoked") else data

    def set(self, event_id: str, data: dict[str, str]) -> None:
        self.vault.put(f"cred:{event_id}", data)

    def clear(self, event_id: str) -> None:
        self.vault.put(f"cred:{event_id}", {"revoked": True})
