#!/usr/bin/env python3
"""Chat do Airbnb/Booking pela rotina do navegador (v1.0.82, Dom 09/10/2026: "Corrige e checa para ver se deu certo").

O sistema poe na fila o que tem de ir ao hospede pelo chat da plataforma (convite com o link, lembretes, respostas da
IA). A rotina do navegador (tarefa agendada, de hora em hora, 8h-21h) le esta lista, manda cada mensagem na caixa do
anfitriao, CONFERE na conversa (o texto aparece la e a caixa ficou vazia) e so entao confirma aqui.

    python ops/entregas_ota.py listar                       # SO LEITURA: o que mandar (JSON), mais antigo primeiro
    python ops/entregas_ota.py descartar-vencidas           # cancela o que passou de 24 h (ou a estadia acabou)
    python ops/entregas_ota.py entregue <chave> [<chave>...]         # conferido na conversa
    python ops/entregas_ota.py nao-consegui <chave> "<motivo curto>"  # gasta uma tentativa (a 5a vira alerta)

Na saida: primeiro nome, quarto, datas, plataforma, codigo da reserva e o texto -- nunca telefone, e-mail ou documento.
"""
from __future__ import annotations

import argparse
import json
import os
import sys
from datetime import datetime, timezone
from typing import Any

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", "api"))


def main(argv: list[str] | None = None, state: Any = None, agora: datetime | None = None) -> int:
    from app.core.record import Actor, ActorKind
    from app.messaging import entrega_ota
    ap = argparse.ArgumentParser()
    ap.add_argument("acao", choices=("listar", "descartar-vencidas", "entregue", "nao-consegui"))
    ap.add_argument("resto", nargs="*")
    a = ap.parse_args(argv)
    if state is None:
        from app.main import build_state
        state = build_state()
    at = agora or datetime.now(timezone.utc)
    ator = Actor(ActorKind.AI, "rotina-navegador-ota")
    if a.acao == "listar":
        print(json.dumps(entrega_ota.a_entregar(state.store, state.outbox, at), ensure_ascii=False, indent=1))
        return 0
    if a.acao == "descartar-vencidas":
        print("vencidas descartadas:", entrega_ota.descartar_vencidas(state, ator=ator, at=at))
        return 0
    if a.acao == "entregue":
        for chave in a.resto:
            print(chave, "->", entrega_ota.confirmar(state, chave, ator=ator, at=at, entregue=True))
        return 0
    if not a.resto:
        print("falta a chave")
        return 2
    print(a.resto[0], "->", entrega_ota.confirmar(state, a.resto[0], ator=ator, at=at, entregue=False,
                                                    motivo=" ".join(a.resto[1:])))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
